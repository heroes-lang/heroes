# Two pthread-locked C counters kept in a record of our own.

extern "counter.h"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i32
    function counter_lock(@c: Counter lent) -> i32
    function counter_unlock(@c: Counter lent) -> i32
    function counter_value(c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

# Sets up the caller's counter in place, so its mutex never moves.
function setup(@c: Counter) -> ()?
    if counter_init(@c) != 0
        return fail(code: "init_failed", msg: "counter_init failed")
    return ok()

# Adds one to the counter's n while holding its lock.
function increment(@c: Counter) -> ()?
    if counter_lock(@c) != 0
        return fail(code: "lock_failed", msg: "counter_lock failed")
    c.n @ c.n + 1
    if counter_unlock(@c) != 0
        return fail(code: "unlock_failed", msg: "counter_unlock failed")
    return ok()

# Gives the counter's value doubled without changing the counter.
function doubled(c: Counter lent) -> i64
    return counter_value(c) * 2

function main()
    s @= Stats(first: Counter(n: 0, rest: zero), second: Counter(n: 0, rest: zero), events: 0)
    setup(@s.first).must()
    setup(@s.second).must()
    i @= 0
    while i < 3
        increment(@s.first).must()
        s.events @ s.events + 1
        i @ i + 1
    j @= 0
    while j < 2
        increment(@s.second).must()
        s.events @ s.events + 1
        j @ j + 1
    print(doubled(s.first))
    print(doubled(s.second))
    print(s.events)
