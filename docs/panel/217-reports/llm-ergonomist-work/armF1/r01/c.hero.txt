# Two pthread-locked counters from counter.h, with an event count.

extern "counter.h" link "pthread"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i32
    function counter_lock(@c: Counter lent) -> i32
    function counter_unlock(@c: Counter lent) -> i32
    function counter_value(c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

# Sets up a new counter in the caller's own place, so its mutex never moves.
function setup_counter(@c: Counter)
    rc = counter_init(@c)
    assert rc == 0

# Adds one to the counter's n while holding its lock.
function increment(@c: Counter)
    locked = counter_lock(@c)
    assert locked == 0
    c.n @ c.n + 1
    unlocked = counter_unlock(@c)
    assert unlocked == 0

# The counter's value doubled; the counter is only read.
function doubled(c: Counter lent) -> i64
    return counter_value(c) * 2

function main()
    stats @= Stats(first: Counter(n: 0, rest: zero), second: Counter(n: 0, rest: zero), events: 0)
    setup_counter(@stats.first)
    setup_counter(@stats.second)
    i @= 0
    while i < 3
        increment(@stats.first)
        stats.events @ stats.events + 1
        i @ i + 1
    j @= 0
    while j < 2
        increment(@stats.second)
        stats.events @ stats.events + 1
        j @ j + 1
    print(doubled(stats.first))
    print(doubled(stats.second))
    print(stats.events)
