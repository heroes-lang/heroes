# Two C counters kept in a record, with an event count.

extern "counter.h"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter) -> i32
    function counter_lock(@c: Counter) -> i32
    function counter_unlock(@c: Counter) -> i32
    function counter_value(c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

# Sets up the caller's counter in place, so its mutex never moves after init.
function setup(@c: Counter) -> ()?
    rc = counter_init(@c)
    if rc != 0
        return fail(code: "counter_init_failed", msg: f"counter_init returned {rc}")
    return ok()

# Adds one to the counter's `n` while holding its lock.
function increment(@c: Counter) -> ()?
    lock_rc = counter_lock(@c)
    if lock_rc != 0
        return fail(code: "counter_lock_failed", msg: f"counter_lock returned {lock_rc}")
    c.n @ c.n + 1
    unlock_rc = counter_unlock(@c)
    if unlock_rc != 0
        return fail(code: "counter_unlock_failed", msg: f"counter_unlock returned {unlock_rc}")
    return ok()

# The counter's value doubled; the counter is only read.
function doubled(c: Counter lent) -> i64
    return counter_value(c) * 2

function main()
    s @= Stats(first: Counter(n: 0, rest: zero), second: Counter(n: 0, rest: zero), events: 0)
    setup(@s.first).must()
    setup(@s.second).must()
    i @= 0
    while i < 3
        increment(@s.first).must()
        s.events @ s.events + 1
        i @ i + 1
    j @= 0
    while j < 2
        increment(@s.second).must()
        s.events @ s.events + 1
        j @ j + 1
    print(doubled(s.first))
    print(doubled(s.second))
    print(s.events)
