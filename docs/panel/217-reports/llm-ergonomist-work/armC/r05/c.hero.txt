# Two mutex-guarded C counters, kept in a record with an event count.

extern "counter.h" link "pthread"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i32
    function counter_lock(@c: Counter lent) -> i32
    function counter_unlock(@c: Counter lent) -> i32
    function counter_value(c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

# Sets up a new counter in the caller's place, so the initialised mutex never moves.
function init_counter(@c: Counter) -> ()?
    if counter_init(@c) != 0
        return fail(code: "init_failed", msg: "counter_init failed")
    return ok()

# Increments the counter's `n` while holding its lock.
function increment(@c: Counter) -> ()?
    if counter_lock(@c) != 0
        return fail(code: "lock_failed", msg: "counter_lock failed")
    c.n @ c.n + 1
    if counter_unlock(@c) != 0
        return fail(code: "unlock_failed", msg: "counter_unlock failed")
    return ok()

# The counter's value, doubled.
function doubled(c: Counter lent) -> i64
    return counter_value(c) * 2

function main()
    stats @= Stats(first: Counter(n: 0, rest: zero), second: Counter(n: 0, rest: zero), events: 0)
    init_counter(@stats.first).must()
    init_counter(@stats.second).must()
    i @= 0
    while i < 3
        increment(@stats.first).must()
        stats.events @ stats.events + 1
        i @ i + 1
    j @= 0
    while j < 2
        increment(@stats.second).must()
        stats.events @ stats.events + 1
        j @ j + 1
    print(doubled(stats.first))
    print(doubled(stats.second))
    print(stats.events)
