# Two C counters kept in a record, incremented under their locks.

extern "counter.h" link "pthread"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i32
    function counter_lock(@c: Counter lent) -> i32
    function counter_unlock(@c: Counter lent) -> i32
    function counter_value(c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

# Ends the program when a pthread call reports an error.
function check(rc: i32, what: str)
    if rc != 0
        print(f"{what} failed: {rc}")
        exit(code: 1)

# Sets up a new counter in the caller's own place, so the mutex never moves.
function setup_counter(@c: Counter)
    check(counter_init(@c), "counter_init")

# Adds one to the counter's n while holding its lock.
function increment(@c: Counter)
    check(counter_lock(@c), "counter_lock")
    c.n @ c.n + 1
    check(counter_unlock(@c), "counter_unlock")

# The counter's value, doubled.
function doubled(c: Counter lent) -> i64
    return counter_value(c) * 2

function main()
    s @= Stats(
        first: Counter(n: 0, rest: zero),
        second: Counter(n: 0, rest: zero),
        events: 0)
    setup_counter(@s.first)
    setup_counter(@s.second)
    i @= 0
    while i < 3
        increment(@s.first)
        s.events @ s.events + 1
        i @ i + 1
    j @= 0
    while j < 2
        increment(@s.second)
        s.events @ s.events + 1
        j @ j + 1
    print(doubled(s.first))
    print(doubled(s.second))
    print(s.events)
