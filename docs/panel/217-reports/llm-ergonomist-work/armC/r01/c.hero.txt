# Two pthread-guarded counters from counter.h, kept in a Stats record.

extern "counter.h"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i32
    function counter_lock(@c: Counter lent) -> i32
    function counter_unlock(@c: Counter lent) -> i32
    function counter_value(c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

# Initialises the caller's counter in place, so its mutex never moves after init.
function setup_counter(@c: Counter) -> ()?
    rc = counter_init(@c)
    if rc != 0
        return fail(code: "init_failed", msg: f"counter_init returned {rc}")
    return ok()

# Adds one to the counter's n while holding its lock.
function increment(@c: Counter) -> ()?
    if counter_lock(@c) != 0
        return fail(code: "lock_failed", msg: "counter_lock failed")
    c.n @ c.n + 1
    if counter_unlock(@c) != 0
        return fail(code: "unlock_failed", msg: "counter_unlock failed")
    return ok()

# The counter's value, doubled.
function doubled(c: Counter lent) -> i64
    return counter_value(c) * 2

function run() -> ()?
    s @= Stats(
        first: Counter(n: 0, rest: zero),
        second: Counter(n: 0, rest: zero),
        events: 0
    )
    setup_counter(@s.first)?
    setup_counter(@s.second)?
    for _ in range(from: 0, to: 3)
        increment(@s.first)?
        s.events @ s.events + 1
    for _ in range(from: 0, to: 2)
        increment(@s.second)?
        s.events @ s.events + 1
    print(doubled(s.first))
    print(doubled(s.second))
    print(s.events)
    return ok()

function main()
    match run()
        .ok _ => _ = 0
        .err e =>
            print(f"error: {e.code}: {e.msg}")
            exit(code: 1)
