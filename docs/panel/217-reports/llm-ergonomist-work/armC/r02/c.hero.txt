# Two mutex-guarded C counters and an event tally.

extern "counter.h"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i32
    function counter_lock(@c: Counter lent) -> i32
    function counter_unlock(@c: Counter lent) -> i32
    function counter_value(c: Counter lent) -> i64

# Two counters and how many events they have counted.
record Stats
    first: Counter
    second: Counter
    events: i64

# Sets up the caller's counter in place, so its mutex never moves once initialised.
function setup(@c: Counter) -> ()?
    rc = counter_init(@c)
    if rc != 0
        return fail(code: "init_failed", msg: f"counter_init returned {rc}")
    return ok()

# Adds one to `n` while holding the counter's lock.
function increment(@c: Counter) -> ()?
    locked = counter_lock(@c)
    if locked != 0
        return fail(code: "lock_failed", msg: f"counter_lock returned {locked}")
    c.n @ c.n + 1
    unlocked = counter_unlock(@c)
    if unlocked != 0
        return fail(code: "unlock_failed", msg: f"counter_unlock returned {unlocked}")
    return ok()

# The counter's value, doubled.
function doubled(c: Counter lent) -> i64
    return counter_value(c) * 2

function run() -> ()?
    s @= Stats(
        first: Counter(n: 0, rest: zero),
        second: Counter(n: 0, rest: zero),
        events: 0)
    setup(@s.first)?
    setup(@s.second)?
    i @= 0
    while i < 3
        increment(@s.first)?
        s.events @ s.events + 1
        i @ i + 1
    j @= 0
    while j < 2
        increment(@s.second)?
        s.events @ s.events + 1
        j @ j + 1
    print(doubled(s.first))
    print(doubled(s.second))
    print(s.events)
    return ok()

function main()
    match run()
        .ok _ => _ = 0
        .err e =>
            print(f"error {e.code}: {e.msg}")
            exit(code: 1)
