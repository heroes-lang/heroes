extern "counter.h"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i32
    function counter_lock(@c: Counter lent) -> i32
    function counter_unlock(@c: Counter lent) -> i32
    function counter_value(@c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

# Sets up a new counter in the caller's place.
function setup(@c: Counter) -> ()?
    rc = counter_init(@c)
    if rc != 0
        return fail(code: "counter_init_failed", msg: f"counter_init returned {rc}")
    return ok()

# Adds one to `n` while holding the counter's lock.
function increment(@c: Counter) -> ()?
    lrc = counter_lock(@c)
    if lrc != 0
        return fail(code: "lock_failed", msg: f"counter_lock returned {lrc}")
    c.n @ c.n + 1
    urc = counter_unlock(@c)
    if urc != 0
        return fail(code: "unlock_failed", msg: f"counter_unlock returned {urc}")
    return ok()

# The counter's value doubled; works on a copy, so the caller's counter is untouched.
function doubled(c: Counter) -> i64
    view @= c
    return counter_value(@view) * 2

function main()
    s @= Stats(first: Counter(rest: zero), second: Counter(rest: zero), events: 0)
    setup(@s.first).must()
    setup(@s.second).must()
    i @= 0
    while i < 3
        increment(@s.first).must()
        s.events @ s.events + 1
        i @ i + 1
    j @= 0
    while j < 2
        increment(@s.second).must()
        s.events @ s.events + 1
        j @ j + 1
    print(doubled(s.first))
    print(doubled(s.second))
    print(s.events)
