extern "counter.h"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i64
    function counter_lock(@c: Counter lent) -> i64
    function counter_unlock(@c: Counter lent) -> i64
    function counter_value(@c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

# Sets up a new counter for the caller.
function new_counter() -> Counter?
    c @= Counter(n: 0, rest: zero)
    rc = counter_init(@c)
    if rc != 0
        return fail(code: "counter_init_failed", msg: f"counter_init returned {rc}")
    return ok(c)

# Increments n while holding the counter's lock.
function increment(@c: Counter) -> ()?
    lr = counter_lock(@c)
    if lr != 0
        return fail(code: "lock_failed", msg: f"counter_lock returned {lr}")
    c.n @ c.n + 1
    ur = counter_unlock(@c)
    if ur != 0
        return fail(code: "unlock_failed", msg: f"counter_unlock returned {ur}")
    return ok()

# The counter's value doubled; the caller's counter is untouched.
function doubled(c: Counter) -> i64
    copy @= c
    v = counter_value(@copy)
    return v * 2

function main()
    s @= Stats(first: new_counter().must(), second: new_counter().must(), events: 0)
    i @= 0
    while i < 3
        increment(@s.first).must()
        s.events @ s.events + 1
        i @ i + 1
    j @= 0
    while j < 2
        increment(@s.second).must()
        s.events @ s.events + 1
        j @ j + 1
    print(doubled(s.first))
    print(doubled(s.second))
    print(s.events)
