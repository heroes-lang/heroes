extern "counter.h"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i32
    function counter_lock(@c: Counter lent) -> i32
    function counter_unlock(@c: Counter lent) -> i32
    function counter_value(@c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

# Sets up a new counter in the caller's own place: `counter_init` zeroes `n`
# and initialises the lock there, so no `Counter` is ever copied.
function setup_counter(@c: Counter) -> ()?
    rc = counter_init(@c)
    if rc != 0
        return fail(code: "counter_init_failed", msg: f"counter_init returned {rc}")
    return ok()

# Increments `n` while holding the counter's lock.
function increment(@c: Counter) -> ()?
    if counter_lock(@c) != 0
        return fail(code: "lock_failed", msg: "counter_lock failed")
    c.n @ c.n + 1
    if counter_unlock(@c) != 0
        return fail(code: "unlock_failed", msg: "counter_unlock failed")
    return ok()

# The counter's value doubled. The counter is lent in place, because a lock
# may not be copied; `counter_value` takes a const pointer and changes nothing.
function doubled(@c: Counter) -> i64
    return counter_value(@c) * 2

function main()
    s @= Stats(first: Counter(rest: zero), second: Counter(rest: zero), events: 0)
    setup_counter(@s.first).must()
    setup_counter(@s.second).must()
    k @= 0
    while k < 3
        increment(@s.first).must()
        s.events @ s.events + 1
        k @ k + 1
    j @= 0
    while j < 2
        increment(@s.second).must()
        s.events @ s.events + 1
        j @ j + 1
    print(doubled(@s.first))
    print(doubled(@s.second))
    print(s.events)
