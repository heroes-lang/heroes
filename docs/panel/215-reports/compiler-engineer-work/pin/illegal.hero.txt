# Compiler-engineer, panel 215: copies a pinned record must refuse.
extern "mx.h"
    record Box tag box partial pinned
        n: i64
    function box_init(@b: Box) -> i32

record Holder
    b: Box
    k: i64

function main()
    m: Box @= Box(n: 0, rest: zero)
    _ = box_init(@m)
    h = Holder(b: Box(n: 5, rest: zero), k: 1)
    h2 = h
    xs = [m]
    same = m.n == 0
    print(same, " ", h2.k, " ", xs.len())
