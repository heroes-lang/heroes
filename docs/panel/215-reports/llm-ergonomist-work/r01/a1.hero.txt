extern "sockpath.h"
    record SockPath tag sock_path
        family: u16
        path: i8[108]
    function sp_send(@p: SockPath lent) -> i32

function main()
    sp @= SockPath(family: 1, rest: zero)
    text = "/tmp/heroes.sock"
    for i in range(from: 0, to: text.len())
        sp.path[i] @ text[i].to_i8().must()
    print(sp_send(@sp))
