extern "counter.h"
    record Counter tag counter partial
        n: i64
    function counter_init(@c: Counter lent) -> i32
    function counter_lock(@c: Counter lent) -> i32
    function counter_unlock(@c: Counter lent) -> i32
    function counter_value(@c: Counter lent) -> i64

record Stats
    first: Counter
    second: Counter
    events: i64

function new_counter() -> Counter
    c: Counter @= Counter(n: 0, rest: zero)
    assert counter_init(@c) == 0
    return c

function doubled(c: Counter) -> i64
    return c.n * 2

function bump(@c: Counter)
    assert counter_lock(@c) == 0
    c.n @ c.n + 1
    assert counter_unlock(@c) == 0

function main()
    s: Stats @= Stats(first: new_counter(), second: new_counter(), events: 0)
    for _ in range(from: 0, to: 3)
        bump(@s.first)
        s.events @ s.events + 1
    for _ in range(from: 0, to: 2)
        bump(@s.second)
        s.events @ s.events + 1
    print(doubled(s.first))
    print(doubled(s.second))
    print(s.events)
